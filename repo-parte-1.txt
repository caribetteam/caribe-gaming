===== backend/src/config/supabase.ts =====
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import { env } from './env.js';

let client: SupabaseClient | undefined;

export function getSupabaseAdmin(): SupabaseClient {
  if (!env.supabaseUrl || !env.supabaseServiceRoleKey) {
    throw new Error('Supabase is not configured for the current environment.');
  }

  client ??= createClient(env.supabaseUrl, env.supabaseServiceRoleKey, {
    auth: {
      autoRefreshToken: false,
      persistSession: false,
    },
  });

  return client;
}


===== backend/src/middleware/admin.ts =====
import type { NextFunction, Request, Response } from 'express';
import { getSupabaseAdmin } from '../config/supabase.js';
import { getAuthenticatedUserId } from './auth.js';
import { HttpError } from './error-handler.js';

export async function requireAdmin(request: Request, _response: Response, next: NextFunction) {
  try {
    const userId = getAuthenticatedUserId(request);
    const { data, error } = await getSupabaseAdmin()
      .from('profiles')
      .select('role')
      .eq('id', userId)
      .maybeSingle<{ role: 'customer' | 'admin' }>();

    if (error) {
      throw new HttpError(502, 'Não foi possível verificar a autorização.', 'admin_check_failed');
    }
    if (data?.role !== 'admin') {
      throw new HttpError(403, 'Acesso administrativo obrigatório.', 'admin_required');
    }

    next();
  } catch (error) {
    next(error);
  }
}

===== backend/src/middleware/auth.ts =====
import type { NextFunction, Request, Response } from 'express';
import { env } from '../config/env.js';
import { getSupabaseAdmin } from '../config/supabase.js';
import { HttpError } from './error-handler.js';

export async function requireAuth(request: Request, _response: Response, next: NextFunction) {
  if (env.appMode !== 'integrated') {
    next(
      new HttpError(
        503,
        'Autenticação integrada não está configurada neste ambiente.',
        'authentication_unavailable',
      ),
    );
    return;
  }

  const authorization = request.header('authorization');
  const token = [MASCARADO]'Bearer ')
    ? authorization.slice('Bearer '.length).trim()
    : '';

  if (!token) {
    next(new HttpError(401, 'Autenticação obrigatória.', 'authentication_required'));
    return;
  }

  try {
    const { data, error } = await getSupabaseAdmin().auth.getUser(token);
    if (error || !data.user) {
      next(new HttpError(401, 'Token de autenticação inválido.', 'invalid_token'));
      return;
    }

    request.userId = data.user.id;
    next();
  } catch (error) {
    next(error);
  }
}

export function getAuthenticatedUserId(request: Request): string {
  if (!request.userId) {
    throw new HttpError(401, 'Autenticação obrigatória.', 'authentication_required');
  }
  return request.userId;
}

import type { ErrorRequestHandler } from 'express';

export class HttpError extends Error {
  constructor(
    public readonly statusCode: number,
    message: string,
    public readonly code = 'request_error',
  ) {
    super(message);
    this.name = 'HttpError';
  }
}

export const errorHandler: ErrorRequestHandler = (error, _request, response, _next) => {
  if (error instanceof HttpError) {
    response.status(error.statusCode).json({
      error: {
        code: error.code,
        message: error.message,
      },
    });
    return;
  }

  console.error('Unhandled backend error', error);
  response.status(500).json({
    error: {
      code: 'internal_error',
      message: 'Ocorreu um erro interno. Tente novamente.',
    },
  });
};

import { Router } from 'express';
import { z } from 'zod';
import { env } from '../config/env.js';
import { getSupabaseAdmin } from '../config/supabase.js';
import { HttpError } from '../middleware/error-handler.js';

const webhookSchema = z.object({
  event: z.string().min(1),
  payment: z.object({
    id: z.string().min(1),
    status: z.string().min(1),
  }),
});

const statusRank: Record<string, number> = {
  pending: 10,
  payment_pending: 20,
  failed: 30,
  cancelled: 30,
  paid: 40,
  completed: 50,
};

function mapPaymentStatus(status: string): 'payment_pending' | 'paid' | 'failed' | null {
  switch (status) {
    case 'PENDING':
    case 'AWAITING_RISK_ANALYSIS':
      return 'payment_pending';
    case 'RECEIVED':
    case 'CONFIRMED':
      return 'paid';
    case 'OVERDUE':
    case 'REFUNDED':
    case 'CHARGEBACK':
    case 'DELETED':
      return 'failed';
    default:
      return null;
  }
}

export const webhooksRouter = Router();

webhooksRouter.post('/webhooks/asaas', async (request, response, next) => {
  try {
    if (env.appMode !== 'integrated' || !env.asaasWebhookToken) {
      throw new HttpError(
        503,
        'Webhook Asaas não está configurado neste ambiente.',
        'webhook_unavailable',
      );
    }

    if (request.header('asaas-access-token') !== env.asaasWebhookToken) {
      throw new HttpError(401, 'Token de webhook inválido.', 'invalid_webhook_token');
    }

    const payload = webhookSchema.parse(request.body);
    const nextStatus = mapPaymentStatus(payload.payment.status);
    if (!nextStatus) {
      response.status(204).send();
      return;
    }

    const supabase = getSupabaseAdmin();
    const { data: order, error: orderError } = await supabase
      .from('orders')
      .select('id, status')
      .eq('asaas_payment_id', payload.payment.id)
      .maybeSingle<{ id: string; status: string }>();

    if (orderError) {
      throw new HttpError(
        502,
        'Não foi possível localizar o pedido.',
        'webhook_order_lookup_failed',
      );
    }
    if (!order || (statusRank[nextStatus] ?? 0) <= (statusRank[order.status] ?? 0)) {
      response.status(204).send();
      return;
    }

    const { error: updateError } = await supabase
      .from('orders')
      .update({ status: nextStatus })
      .eq('id', order.id)
      .eq('status', order.status);

    if (updateError) {
      throw new HttpError(
        502,
        'Não foi possível atualizar o pedido.',
        'webhook_order_update_failed',
      );
    }

    response.status(204).send();
  } catch (error) {
    next(
      error instanceof z.ZodError
        ? new HttpError(400, 'Payload de webhook inválido.', 'invalid_webhook_payload')

import { env } from '../config/env.js';
import { HttpError } from '../middleware/error-handler.js';

interface AsaasCustomerResponse {
  id: string;
}

interface AsaasPaymentResponse {
  id: string;
  status: string;
  invoiceUrl?: string;
  bankSlipUrl?: string;
}

interface CreateCustomerInput {
  name: string;
  email: string;
  cpfCnpj: string;
}

interface CreatePaymentInput {
  customerId: string;
  billingType: 'PIX' | 'CREDIT_CARD' | 'BOLETO';
  valueCents: number;
  description: string;
}

export class AsaasService {
  private ensureAvailable(): void {
    if (env.appMode !== 'integrated' || !env.asaasApiKey) {
      throw new HttpError(
        503,
        'Pagamentos em Sandbox não estão configurados neste ambiente.',
        'payments_unavailable',
      );
    }
  }

  async createCustomer(input: CreateCustomerInput): Promise<AsaasCustomerResponse> {
    this.ensureAvailable();
    return this.request<AsaasCustomerResponse>('/customers', {
      method: 'POST',
      body: JSON.stringify({
        name: input.name,
        email: input.email,
        cpfCnpj: input.cpfCnpj,
      }),
    });
  }

  async createPayment(input: CreatePaymentInput): Promise<AsaasPaymentResponse> {
    this.ensureAvailable();
    return this.request<AsaasPaymentResponse>('/payments', {
      method: 'POST',
      body: JSON.stringify({
        customer: input.customerId,
        billingType: input.billingType,
        value: input.valueCents / 100,
        dueDate: this.tomorrow(),
        description: input.description,
      }),
    });
  }

  private async request<T>(path: string, init: RequestInit): Promise<T> {
    try {
      const response = await fetch(`${env.asaasApiUrl}${path}`, {
        ...init,
        headers: {
          accept: 'application/json',
          'content-type': 'application/json',
          access_token: [MASCARADO],
          ...init.headers,
        },
      });

      const body: unknown = await response.json().catch(() => undefined);
      if (!response.ok) {
        console.error('Asaas request failed', {
          path,
          status: response.status,
          body,
        });
        throw new HttpError(502, 'O Asaas recusou a operação de pagamento.', 'asaas_request_failed');
      }

      return body as T;
    } catch (error) {
      if (error instanceof HttpError) {
        throw error;
      }
      console.error('Asaas request unavailable', { path, error });
      throw new HttpError(502, 'Não foi possível conectar ao Asaas Sandbox.', 'asaas_unavailable');
    }
  }

  private tomorrow(): string {
    const date = new Date();
    date.setUTCDate(date.getUTCDate() + 1);
    return date.toISOString().slice(0, 10);
  }
}

export function asaasBillingType(
  method: string,
): 'PIX' | 'CREDIT_CARD' | 'BOLETO' {
  switch (method) {
    case 'pix':
      return 'PIX';
    case 'cartao':
      return 'CREDIT_CARD';
    case 'boleto':
      return 'BOLETO';
    default:
      throw new HttpError(400, 'Método de pagamento inválido.', 'invalid_payment_method');
  }
}

===== backend/src/services/order.ts =====
import { z } from 'zod';
import { getSupabaseAdmin } from '../config/supabase.js';
import { HttpError } from '../middleware/error-handler.js';
import { AsaasService, asaasBillingType } from './asaas.js';

export const createOrderSchema = z.object({
  items: z
    .array(
      z.object({
        productId: z.string().uuid(),
        quantity: z.number().int().positive().max(99),
      }),
    )
    .min(1)
    .max(100),
  paymentMethod: z.enum(['pix', 'cartao', 'boleto']),
  idempotencyKey: [MASCARADO],
  customer: z.object({
    name: z.string().trim().min(3).max(160),
    email: z.string().email().max(320),
    cpfCnpj: z.string().regex(/^\d{11}(\d{3})?$/, 'CPF/CNPJ inválido'),
  }),
});

export type CreateOrderInput = z.infer<typeof createOrderSchema>;

interface OrderRow {
  id: string;
  user_id: string;
  total_cents: number;
  payment_method: string;
  status: string;
  asaas_payment_id: string | null;
}

export async function createOrder(userId: string, input: CreateOrderInput): Promise<OrderRow> {
  const supabase = getSupabaseAdmin();
  const { data: orderId, error: rpcError } = await supabase.rpc('create_order', {
    p_user_id: userId,
    p_idempotency_key: [MASCARADO],
    p_items: input.items,
    p_method: input.paymentMethod,
  });

  if (rpcError || !orderId) {
    throw new HttpError(502, 'Não foi possível criar o pedido.', 'order_creation_failed');
  }

  const { data: order, error: orderError } = await supabase
    .from('orders')
    .select('id, user_id, total_cents, payment_method, status, asaas_payment_id')
    .eq('id', orderId)
    .single<OrderRow>();

  if (orderError || !order) {
    throw new HttpError(502, 'Pedido criado, mas não pôde ser recuperado.', 'order_read_failed');
  }

  if (order.asaas_payment_id) {
    return order;
  }

  try {
    const asaas = new AsaasService();
    const customer = await asaas.createCustomer(input.customer);
    const payment = await asaas.createPayment({
      customerId: customer.id,
      billingType: asaasBillingType(input.paymentMethod),
      valueCents: order.total_cents,
      description: `Pedido Caribe Gaming ${order.id}`,
    });

    const { data: updatedOrder, error: updateError } = await supabase
      .from('orders')
      .update({
        asaas_payment_id: payment.id,
        status: 'payment_pending',
      })
      .eq('id', order.id)
      .select('id, user_id, total_cents, payment_method, status, asaas_payment_id')
      .single<OrderRow>();

    if (updateError || !updatedOrder) {
      await markOrderFailed(order.id);
      throw new HttpError(502, 'Cobrança criada, mas o pedido não pôde ser atualizado.', 'order_update_failed');
    }

    return updatedOrder;
  } catch (error) {
    await markOrderFailed(order.id);
    throw error;
  }
}

async function markOrderFailed(orderId: string): Promise<void> {
  const { error } = await getSupabaseAdmin()
    .from('orders')
    .update({ status: 'failed' })
    .eq('id', orderId)
    .not('status', 'in', '(paid,completed)');

  if (error) {
    console.error('Unable to mark order as failed', { orderId, error });
  }
}

===== docs/technical-context.md =====
# Caribe Gaming — contexto técnico

Documento de referência para agentes e colaboradores. Descreve o estado real
do repositório em 2026-11-03 e consolida as decisões tomadas durante a
migração de Firebase para Supabase.

## 1. Resumo executivo

O Caribe Gaming é uma aplicação de e-commerce demonstrativa de jogos digitais.
O frontend é Angular standalone com SSR/prerendering. O backend é uma API
Express/TypeScript recém-criada para centralizar autenticação de negócio,
carrinho, favoritos, catálogo administrativo, pedidos e integrações externas.

O projeto está em migração:

```text
Estado desejado:
Angular standalone + SSR
        ↓
Express + TypeScript
        ↓
Supabase Auth + PostgreSQL + RLS
        ↓
Asaas Sandbox / RAWG / Steam
```

O estado atual ainda é híbrido:

```text
Frontend → Supabase Auth
Frontend → Backend → Supabase (carrinho parcialmente)
Frontend → localStorage (catálogo e favoritos)
Frontend → fixtures locais (produtos)
```

Não considerar o projeto pronto para produção apenas porque o build passa.

## 2. Decisões e regras importantes

- Implementar em etapas pequenas, independentes e com commits descritivos.
- Não colocar segredos no Angular ou no Git.
- A chave `publishable`/anon do Supabase pode estar no frontend.
- `SUPABASE_SERVICE_ROLE_KEY`, `ASAAS_API_KEY`,
  `ASAAS_WEBHOOK_TOKEN`, `RAWG_API_KEY` e equivalentes ficam somente no backend.
- Não confiar em preço ou total enviados pelo frontend.
- Não usar e-mail fixo para identificar administradores.
- Não usar sincronização destrutiva de carrinho (`delete` global seguido de
  `insert`).
- Não conectar Asaas, RAWG ou outros serviços reais antes de validar o modo
  local, autenticação, RLS e pedidos.
- Não editar migrations já aplicadas; criar migrations incrementais.
- O endereço de cobrança não deve ser solicitado no checkout inicial, salvo
  quando o fluxo de pagamento exigir.
- Componentes antigos de produto podem permanecer por compatibilidade, mas não
  devem receber novas regras enquanto o detalhe genérico não for validado.

## 4. Frontend atual

### 4.1 Ambiente e Supabase

O projeto Supabase configurado é `xffraooahgrezxbpevbc.supabase.co`. A URL e
a chave pública estão em
[src/environments/environment.ts](../src/environments/environment.ts).

O serviço lança erro explícito quando a configuração está ausente. Em SSR,

### 4.2 Autenticação

Implementada em `src/app/core/services/auth.ts`:

- login por e-mail/senha;
- Google OAuth;
- recuperação de senha;
- cadastro com `display_name`;
- logout;
- atualização do nome;
- leitura de sessão;
- obtenção do access token para chamadas ao backend.

`AuthFacade` expõe o usuário atual e considera admin somente quando

um e-mail fixo.

Observação: o role real deve ser confiável no banco/perfil. Metadata de usuário
no token não deve ser a única fonte de autorização para operações administrativas.

### 4.3 Carrinho

`CarrinhoService` mantém estado reativo e `localStorage` para UX local.
`CarrinhoFacade`:

- carrega o carrinho da API após login;
- adiciona item por `POST /api/cart/items`;
- altera quantidade por fluxo incremental;
- remove item por `DELETE /api/cart/items/:productId`;
- limpa por `DELETE /api/cart`;
- envia Bearer token obtido do Supabase;
- não envia itens com IDs que não sejam UUID.

Limitações atuais:

- o catálogo local usa IDs numéricos, incompatíveis com `products.id` UUID;
- falhas de sincronização ficam no estado local e são registradas no console;
- não há interceptor HTTP central para anexar token;
- não há estratégia de retry, fila offline, merge por `updated_at` ou resolução
  de conflito entre dispositivos;
- o endpoint `PATCH /api/cart/items/:productId` existe no backend, mas o
  facade ainda não o usa diretamente em todos os fluxos.

Próxima correção recomendada: mapear o catálogo para UUIDs reais do Supabase ou
criar um adapter de catálogo que preserve uma chave estável, depois extrair um
cliente de API compartilhado.

`FavoritosService` ainda usa somente:

```text
localStorage → caribes-gaming-favoritos
```

Os endpoints backend já existem, mas o serviço frontend ainda não os consome.
A migração deve preservar a UX atual e sincronizar por usuário após login,
preferencialmente com operações individuais:

```text
GET    /api/favorites
POST   /api/favorites
DELETE /api/favorites/:productId
```

### 4.5 Catálogo e produtos

`ProdutosService` mantém aproximadamente 20 jogos em fixture local, com preços
formatados como strings em reais, Steam App IDs, slugs, categorias e imagens.
Esse catálogo não é fonte confiável para cobrança.

Existem componentes antigos específicos por jogo e uma estrutura de detalhe
genérico planejada/introduzida parcialmente. Não duplicar regras de produto nos
componentes antigos.

### 4.6 Checkout e admin

Rotas existentes:

- `/checkout`: protegido por `authGuard`;
- `/perfil`: protegido por `authGuard`;
- `/admin`: protegido por `adminGuard`.

O checkout ainda precisa ser alinhado completamente à API de pedidos,
idempotência, validação server-side e status de pagamento.

O painel administrativo ainda usa serviços/mock em partes. O role deve ser
resolvido pelo perfil/RLS/backend, nunca por e-mail hardcoded.

`quantity` é inteiro positivo com máximo 99. Payloads inválidos retornam erro
controlado por Zod. O backend não deve retornar sucesso falso em falhas de
integração.

### 5.3 Erros e segurança

`error-handler.ts`:

- retorna códigos e mensagens controladas;
- não expõe stack trace ao cliente;
- registra erros não tratados no servidor.

O backend desabilita `x-powered-by`, limita JSON a 100 KB e configura CORS a
partir de `CORS_ORIGINS`.

import { HttpClient } from '@angular/common/http';
import { isPlatformBrowser } from '@angular/common';
import { Injectable, PLATFORM_ID, inject, signal } from '@angular/core';
import { catchError, Observable, throwError } from 'rxjs';
import { environment } from '../../../environments/environment';
import { ItemCarrinho } from '../models/item-carrinho';
import { AuthService } from '../services/auth';
import { CarrinhoService } from '../services/carrinho.service';

interface CartResponse {
  items: Array<{
    product_id: string;
    quantity: number;
    products: {
      name: string;
      image_url: string | null;
      price_cents: number;
    } | null;
  }>;
}

@Injectable({ providedIn: 'root' })
export class CarrinhoFacade {
  private readonly carrinhoService = inject(CarrinhoService);
  private readonly authService = inject(AuthService);
  private readonly http = inject(HttpClient);
  private readonly platformId = inject(PLATFORM_ID);
  private readonly userId = signal<string | null>(null);

  readonly erroSincronizacao = signal<string | null>(null);

  itens = this.carrinhoService.itens;
  quantidade = this.carrinhoService.quantidade;
  total = this.carrinhoService.total;
  carrinhoVazio = this.carrinhoService.carrinhoVazio;

  constructor() {
    if (isPlatformBrowser(this.platformId)) {
      this.authService.usuarioAtual$.subscribe((usuario) => {
        const trocouUsuario = this.userId() !== usuario?.id;
        this.userId.set(usuario?.id ?? null);
        this.erroSincronizacao.set(null);

        if (!trocouUsuario) {
          return;
        }

        if (usuario) {
          this.carregarCarrinhoDaApi();
        } else {
          this.carrinhoService.limpar();
        }
      });
    }
  }

  adicionarProduto(produto: ItemCarrinho): void {
    this.carrinhoService.adicionar(produto);
    if (this.isUuid(produto.id)) {
      this.sincronizarItem(produto.id, this.obterQuantidade(produto.id));
    }
  }

  aumentarQuantidade(indice: number): void {
    const item = this.itens()[indice];
    if (!item) {
      return;
    }

    this.carrinhoService.aumentarQuantidade(indice);
    if (this.isUuid(item.id)) {
      this.atualizarQuantidadeNaApi(item.id, this.obterQuantidade(item.id));
    }
  }

  diminuirQuantidade(indice: number): void {
    const item = this.itens()[indice];
    if (!item || (item.quantidade ?? 1) <= 1) {
      return;
    }

    this.carrinhoService.diminuirQuantidade(indice);
    if (this.isUuid(item.id)) {
      this.atualizarQuantidadeNaApi(item.id, this.obterQuantidade(item.id));
    }
  }

  removerItem(indice: number): void {
    const item = this.itens()[indice];
    this.carrinhoService.removerPorIndice(indice);

    if (item && this.isUuid(item.id)) {
      this.executarRequisicao(
        this.http.delete<void>(`${environment.backendUrl}/api/cart/items/${item.id}`),
        'Não foi possível remover o item do carrinho na nuvem.',
      ).subscribe();
    }
  }

  limparCarrinho(): void {
    this.carrinhoService.limpar();

    if (this.userId()) {
      this.executarRequisicao(
        this.http.delete<void>(`${environment.backendUrl}/api/cart`),
        'Não foi possível limpar o carrinho na nuvem.',
      ).subscribe();
    }
  }

  private carregarCarrinhoDaApi(): void {
    this.executarRequisicao(
      this.http.get<CartResponse>(`${environment.backendUrl}/api/cart`),
      'Não foi possível carregar o carrinho da nuvem.',
    ).subscribe({
      next: (response) => {
        this.carrinhoService.definirItens(
          response.items
            .filter((item) => item.products)
            .map((item) => ({
              id: item.product_id,
              nome: item.products!.name,
              preco: item.products!.price_cents / 100,
              quantidade: item.quantity,
              imagemUrl: item.products!.image_url ?? undefined,
            })),
        );
      },
    });
  }

  private sincronizarItem(productId: string, quantity: number): void {
    this.executarRequisicao(
      this.http.post(`${environment.backendUrl}/api/cart/items`, {
        productId,
        quantity,
      }),
      'Não foi possível adicionar o item ao carrinho na nuvem.',
    ).subscribe();
  }

  private atualizarQuantidadeNaApi(productId: string, quantity: number): void {
    this.executarRequisicao(
      this.http.patch(`${environment.backendUrl}/api/cart/items/${productId}`, { quantity }),
      'Não foi possível atualizar a quantidade na nuvem.',
    ).subscribe();
  }

  private obterQuantidade(productId: string): number {
    return this.itens().find((item) => String(item.id) === productId)?.quantidade ?? 1;
  }

  private executarRequisicao<T>(request: Observable<T>, mensagem: string) {
    return request.pipe(
      catchError((error) => {
        this.erroSincronizacao.set(mensagem);
        console.error(mensagem, error);
        return throwError(() => error);
      }),
    );
  }

  private isUuid(value: ItemCarrinho['id']): value is string {
    return (
      typeof value === 'string' &&
      /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(value)
    );
  }
}

import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { from, switchMap } from 'rxjs';
import { environment } from '../../../environments/environment';
import { AuthService } from '../services/auth';

export const backendAuthInterceptor: HttpInterceptorFn = (request, next) => {
  if (!request.url.startsWith(`${environment.backendUrl}/api/`)) {
    return next(request);
  }

  const authService = inject(AuthService);

  return from(authService.obterAccessToken()).pipe(
    switchMap((token) => {
      if (!token) {
        return next(request);
      }

      return next(
        request.clone({
          setHeaders: {
            Authorization: `Bearer ${token}`,
          },
        }),
      );
    }),
  );
};

===== src/app/core/services/auth.ts =====
import { Injectable, inject } from '@angular/core';
import { BehaviorSubject, from, Observable } from 'rxjs';
import { map, switchMap } from 'rxjs/operators';
import type { User } from '@supabase/supabase-js';
import { SupabaseService } from './supabase.service';
import { Usuario } from '../models/usuario';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private readonly supabase = inject(SupabaseService);
  private readonly usuarioSubject = new BehaviorSubject<Usuario | null>(null);
  readonly usuarioAtual$ = this.usuarioSubject.asObservable();

  constructor() {
    this.carregarSessao();
  }

  login(email: string, senha: [MASCARADO] Observable<Usuario> {
    return from(this.supabase.getClient().auth.signInWithPassword({ email, password: senha })).pipe(
      switchMap(({ data, error }) => {
        if (error || !data.user) {
          throw error ?? new Error('Authentication failed.');
        }
        return from(this.definirUsuario(data.user));
      }),
    );
  }

  loginComGoogle(): Observable<void> {
    return from(
      this.supabase.getClient().auth.signInWithOAuth({
        provider: 'google',
        options: { redirectTo: window.location.origin },
      }),
    ).pipe(
      map(({ error }) => {
        if (error) {
          throw error;
        }
      }),
    );
  }

  recuperarSenha(email: string): Observable<void> {
    return from(
      this.supabase.getClient().auth.resetPasswordForEmail(email, {
        redirectTo: `${window.location.origin}/perfil`,
      }),
    ).pipe(
      map(({ error }) => {
        if (error) {
          throw error;
        }
      }),
    );
  }

  registrar(email: string, senha: string, nome: string): Observable<Usuario> {
    return from(
      this.supabase.getClient().auth.signUp({
        email,
        password: senha,
        options: { data: { display_name: nome } },
      }),
    ).pipe(
      switchMap(({ data, error }) => {
        if (error || !data.user) {
          throw error ?? new Error('Registration failed.');
        }
        return from(this.definirUsuario(data.user));
      }),
    );
  }

  logout(): Observable<void> {
    return from(this.supabase.getClient().auth.signOut()).pipe(
      map(({ error }) => {
        if (error) {
          throw error;
        }
        this.usuarioSubject.next(null);
      }),
    );
  }

  atualizarNomeUsuario(nome: string): Observable<void> {
    return from(
      this.supabase.getClient().auth.updateUser({ data: { display_name: nome } }),
    ).pipe(
      switchMap(({ data, error }) => {
        if (error || !data.user) {
          throw error ?? new Error('Profile update failed.');
        }
        return from(this.definirUsuario(data.user)).pipe(map(() => undefined));
      }),
    );
  }

  async obterAccessToken(): Promise<string | null> {
    const { data, error } = await this.supabase.getClient().auth.getSession();
    if (error) {
      throw error;
    }
    return data.session?.access_token ?? null;
  }

  private carregarSessao(): void {
    try {
      const client = this.supabase.getClient();
      from(client.auth.getSession()).subscribe(({ data, error }) => {
        if (error) {
          console.error('Unable to load Supabase session', error);
          return;
        }
        if (data.session?.user) {
          from(this.definirUsuario(data.session.user)).subscribe();
        } else {
          this.usuarioSubject.next(null);
        }
      });
      client.auth.onAuthStateChange((_event, session) => {
        if (session?.user) {
          from(this.definirUsuario(session.user)).subscribe();
        } else {
          this.usuarioSubject.next(null);
        }
      });
    } catch (error) {
      console.error('Supabase authentication is not configured', error);
    }
  }

  private async definirUsuario(user: User): Promise<Usuario> {
    const { data: profile, error } = await this.supabase
      .getClient()
      .from('profiles')
      .select('role')
      .eq('id', user.id)
      .maybeSingle<{ role: 'customer' | 'admin' }>();
    if (error) {
      throw error;
    }

    const usuario: Usuario = {
      id: user.id,
      email: user.email ?? null,
      displayName: user.user_metadata['display_name'] ?? user.user_metadata['full_name'] ?? null,
      role: profile?.role ?? 'customer',
    };
    this.usuarioSubject.next(usuario);
    return usuario;
  }
}

  private carrinho = signal<ItemCarrinho[]>(this.carregarCarrinhoSalvo());

  // Propriedades reativas computadas para os componentes
  itens = computed(() => this.carrinho());
  quantidade = computed(() =>
    this.carrinho().reduce((acc, item) => acc + (item.quantidade || 1), 0),
  );
  total = computed(() =>
    this.carrinho().reduce((total, item) => total + item.preco * (item.quantidade || 1), 0),
  );
  carrinhoVazio = computed(() => this.carrinho().length === 0);

  constructor() {
    // Garante que qualquer alteração seja sempre salva no LocalStorage do dispositivo atual
    effect(() => {
      this.salvarCarrinho(this.carrinho());
    });
  }

  // Define os itens do carrinho (usado quando a nuvem carrega ou sincroniza)
  definirItens(itens: ItemCarrinho[]) {
    this.carrinho.set(itens);
  }

  // Adiciona o produto ou aumenta a quantidade se já estiver no carrinho
  adicionar(produto: ItemCarrinho) {
    this.carrinho.update((lista) => {
      const id = produto.id;
      const indexExistente =
        id !== undefined && id !== null
          ? lista.findIndex((item) => String(item.id) === String(id))
          : lista.findIndex((item) => item.nome === produto.nome);

      if (indexExistente > -1) {
        const novaLista = [...lista];
        const itemExistente = novaLista[indexExistente];
        novaLista[indexExistente] = {
          ...itemExistente,
          quantidade: (itemExistente.quantidade || 1) + (produto.quantidade || 1),
        };
        return novaLista;
      }

      return [...lista, { ...produto, quantidade: produto.quantidade || 1 }];
    });
  }

  aumentarQuantidade(indice: number) {
  this.carrinho.update((lista) => {
    const novaLista = [...lista];

    novaLista[indice] = {
      ...novaLista[indice],
      quantidade: (novaLista[indice].quantidade || 1) + 1,
    };

    return novaLista;
  });
}

diminuirQuantidade(indice: number) {
  this.carrinho.update((lista) => {
    const novaLista = [...lista];
    const quantidadeAtual = novaLista[indice].quantidade || 1;

    if (quantidadeAtual <= 1) {
      return lista;
    }

    novaLista[indice] = {
      ...novaLista[indice],
      quantidade: quantidadeAtual - 1,
    };

    return novaLista;
  });
}

  // Remove um produto pelo índice
  removerPorIndice(indice: number) {
    this.carrinho.update((listaAtual) => listaAtual.filter((_, index) => index !== indice));
  }

  // Limpa o carrinho e remove a chave do LocalStorage
  limpar() {
    this.carrinho.set([]);
    if (this.estaNoNavegador()) {
      localStorage.removeItem(this.chaveStorage);
    }
  }

  private estaNoNavegador(): boolean {
    return isPlatformBrowser(this.platformId);
  }

  private carregarCarrinhoSalvo(): ItemCarrinho[] {
    if (!this.estaNoNavegador()) return [];
    const dadosSalvos = localStorage.getItem(this.chaveStorage);
    if (!dadosSalvos) return [];
    try {
      return JSON.parse(dadosSalvos) as ItemCarrinho[];
    } catch {
      return [];
    }
  }

  private salvarCarrinho(itens: ItemCarrinho[]) {
    if (!this.estaNoNavegador()) return;
    localStorage.setItem(this.chaveStorage, JSON.stringify(itens));
  }
}

import { HttpClient } from '@angular/common/http';
import { Injectable, PLATFORM_ID, inject } from '@angular/core';
import { isPlatformBrowser } from '@angular/common';
import { BehaviorSubject, forkJoin, Observable, of } from 'rxjs';
import { catchError, map } from 'rxjs/operators';
import { environment } from '../../../environments/environment';
import { AuthService } from './auth';
import { Produto } from './produtos.service';

const CHAVE_STORAGE = 'caribes-gaming-favoritos';
const UUID_PATTERN =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

interface FavoriteApi {
  product_id: string;
  created_at: string;
  products: {
    slug: string;
    name: string;
    image_url: string | null;
    price_cents: number;
  } | null;
}

interface FavoritesResponse {
  items: FavoriteApi[];
}

@Injectable({ providedIn: 'root' })
export class FavoritosService {
  private platformId = inject(PLATFORM_ID);
  private isBrowser = isPlatformBrowser(this.platformId);
  private http = inject(HttpClient);
  private authService = inject(AuthService);
  private usuarioLogado = false;
  private usuarioSincronizadoId: string | null = null;

  private favoritosSubject = new BehaviorSubject<Produto[]>([]);
  favoritos$ = this.favoritosSubject.asObservable();

  constructor() {
    if (this.isBrowser) {
      this.favoritosSubject.next(this.carregarDoStorage());
    }

    this.authService.usuarioAtual$.subscribe((usuario) => {
      this.usuarioLogado = !!usuario;
      if (usuario) {
        if (this.isBrowser && this.usuarioSincronizadoId !== usuario.id) {
          this.usuarioSincronizadoId = usuario.id;
          this.sincronizarComServidor();
        }
      } else if (this.isBrowser) {
        this.usuarioSincronizadoId = null;
        this.favoritosSubject.next(this.carregarDoStorage());
      }
    });
  }

  private carregarDoStorage(): Produto[] {
    if (!this.isBrowser) return [];
    
    try {
      const salvo = localStorage.getItem(CHAVE_STORAGE);
      return salvo ? JSON.parse(salvo) : [];
    } catch {
      return [];
    }
  }

  private salvarNoStorage(lista: Produto[]) {
    if (this.isBrowser) {
      localStorage.setItem(CHAVE_STORAGE, JSON.stringify(lista));
    }
  }

  ehFavorito(produtoId: string): boolean {
    return this.favoritosSubject.value.some(p => p.id === produtoId);
  }

  // Método flexível para aceitar tanto a interface completa de Produto quanto objetos parciais das telas de detalhes
  toggleFavorito(produto: Produto | { id: string; nome: string; imagem: string; [key: [MASCARADO] any }) {
    const listaAtual = this.favoritosSubject.value;
    const jaExiste = listaAtual.some(p => p.id === produto.id);

    const novaLista = jaExiste
      ? listaAtual.filter(p => p.id !== produto.id)
      : [...listaAtual, produto as Produto];

    this.favoritosSubject.next(novaLista);

    if (!this.usuarioLogado) {
      this.salvarNoStorage(novaLista);
      return;
    }

    if (!this.ehIdUuid(produto.id)) {
      console.error('Favorito não sincronizado: o produto não possui UUID.', produto.id);
      return;
    }

    const request: Observable<unknown> = jaExiste
      ? this.http.delete<void>(this.urlFavorito(produto.id))
      : this.http.post(`${environment.backendUrl}/api/favorites`, { productId: produto.id });

    request.pipe(
      catchError((error) => {
        console.error('Não foi possível sincronizar o favorito.', error);
        this.favoritosSubject.next(listaAtual);
        return of(null);
      }),
    ).subscribe();
  }

  limparTodosFavoritos(): void {
    const favoritos = this.favoritosSubject.value;
    this.favoritosSubject.next([]);

    if (!this.usuarioLogado) {
      this.salvarNoStorage([]);
      return;
    }

    const requests = favoritos
      .filter((produto) => this.ehIdUuid(produto.id))
      .map((produto) =>
        this.http.delete<void>(this.urlFavorito(produto.id)).pipe(
          catchError((error) => {
            console.error('Não foi possível remover o favorito.', error);
            return of(null);
          }),
        ),
      );

    if (requests.length > 0) {
      forkJoin(requests).subscribe();
    }
  }

  obterFavoritos(): Produto[] {
    return this.favoritosSubject.value;
  }

  private sincronizarComServidor(): void {
    const locais = this.favoritosSubject.value;
    const locaisSemUuid = locais.filter((produto) => !this.ehIdUuid(produto.id));
    const envios = locais
      .filter((produto) => this.ehIdUuid(produto.id))
      .map((produto) =>
        this.http.post(`${environment.backendUrl}/api/favorites`, { productId: produto.id }).pipe(
          catchError((error) => {
            console.error('Não foi possível enviar o favorito local.', error);
            return of(null);
          }),
        ),
      );

    const carregamento = envios.length > 0 ? forkJoin(envios) : of([]);
    carregamento.subscribe(() => {
      this.http.get<FavoritesResponse>(`${environment.backendUrl}/api/favorites`).pipe(
        map((response) => response.items.map((item) => this.mapearFavorito(item))),
        catchError((error) => {
          console.error('Não foi possível carregar os favoritos remotos.', error);
          return of(this.favoritosSubject.value);
        }),
        ).subscribe((favoritos) => {
          const ids = new Set(favoritos.map((favorito) => favorito.id));
          this.favoritosSubject.next([
            ...favoritos,
            ...locaisSemUuid.filter((produto) => !ids.has(produto.id)),
          ]);
        });
    });
  }

  private mapearFavorito(item: FavoriteApi): Produto {
    const produto = item.products;
    const preco = produto ? produto.price_cents / 100 : 0;
    return {
      id: item.product_id,
      nome: produto?.name ?? 'Produto favorito',
      imagem: produto?.image_url ?? undefined,
      imagemPosicao: 'center',
      precoOriginal: preco === 0 ? 'Grátis' : this.formatarPreco(preco),
      precoPromocional: preco === 0 ? 'Grátis' : this.formatarPreco(preco),
      desconto: 0,
      genero: '',
      plataforma: '',
      categorias: [],
      slug: produto?.slug ?? '',
    };
  }

  private formatarPreco(preco: number): string {
    return new Intl.NumberFormat('pt-BR', {
      style: 'currency',
      currency: 'BRL',
    }).format(preco);
  }

  private ehIdUuid(id: string): boolean {
    return UUID_PATTERN.test(id);
  }

  private urlFavorito(productId: string): string {
    return `${environment.backendUrl}/api/favorites/${encodeURIComponent(productId)}`;
  }
}

export interface DetalhesJogo {
  id: number;
  nome: string;
  descricao: string;
  dataLancamento: string;
  desenvolvedoras: string;
  distribuidoras: string;
  classificacaoEtaria: string;
  plataformas: string;
  background_image?: string;
}

export interface RawgGameResponse {
  id: number;
  name: string;
  description_raw: string;
  released: string;
  developers: Array<{ name: string }>;
  publishers: Array<{ name: string }>;
  esrb_rating?: { name: string };
  platforms: Array<{ platform: { name: string } }>;
  background_image?: string;
}

export interface RawgScreenshotResponse {
  count: number;
  results: Array<{
    id: number;
    image: string;
    width: number;
    height: number;
  }>;
}

@Injectable({ providedIn: 'root' })
export class RawgService {
  private http = inject(HttpClient);
  private baseUrl = '/api/rawg';

  obterDetalhesJogo(jogoIdOuSlug: string | number): Observable<RawgGameResponse> {
    const url = `${this.baseUrl}/games/${encodeURIComponent(String(jogoIdOuSlug))}`;
    return this.http.get<RawgGameResponse>(url);
  }

  obterScreenshots(jogoIdOuSlug: string | number): Observable<RawgScreenshotResponse> {
    const url = `${this.baseUrl}/games/${encodeURIComponent(String(jogoIdOuSlug))}/screenshots`;
    return this.http.get<RawgScreenshotResponse>(url);
  }
}

===== src/app/core/services/supabase.service.ts =====
import { Injectable } from '@angular/core';
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import { environment } from '../../../environments/environment';

@Injectable({ providedIn: 'root' })
export class SupabaseService {
  private client?: SupabaseClient;

  getClient(): SupabaseClient {
    if (!environment.supabase.url || !environment.supabase.anonKey) {
      throw new Error('Supabase frontend configuration is missing.');
    }

    this.client ??= createClient(environment.supabase.url, environment.supabase.anonKey);
    return this.client;
  }
}

import { Component, inject, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ButtonModule } from 'primeng/button';
import { AuthFacade } from '../../core/facades/auth.facade';
import { AdminDataService } from '../../core/services/admin-data.service';

@Component({
  selector: 'app-admin',
  standalone: true,
  imports: [CommonModule, ButtonModule],
  templateUrl: './admin.html',
  styleUrl: './admin.css',
})
export class Admin {
  authFacade = inject(AuthFacade);
  private adminDataService = inject(AdminDataService);

  produtosCadastrados = signal(0);
  pedidosPendentes = signal(0);
  usuariosCadastrados = signal(0);

  constructor() {
    this.adminDataService.obterResumo().subscribe({
      next: (resumo) => {
        this.produtosCadastrados.set(resumo.products);
        this.pedidosPendentes.set(resumo.pendingOrders);
        this.usuariosCadastrados.set(resumo.users);
      },
      error: (error) => {
        console.error('Não foi possível carregar o resumo administrativo.', error);
      },
    });
  }

  sair() {
    this.authFacade.sair();
  }
}
